A — flywheel: output.rs → hacker-night truecolor (3-tier degrade)
// theme: semantic slot → (truecolor RGB, 256 idx, 16 ANSI code)
pub enum Slot { BgBase, BgSurface, FgDefault, FgMuted,
  FgEmphasis, AccentCyan, AccentLav, Error, Warn, Success, Info }

// hacker-night exacts
fn rgb(s: Slot) -> (u8,u8,u8) {
  match s {
    BgBase => (0x08,0x08,0x0e), // void
    FgMuted => (0x56,0x5f,0x89),
    AccentCyan => (0x0d,0xb9,0xd7), // vivid cyan
    AccentLav => (0xbb,0x9a,0xf7),
    Error => (0xf7,0x76,0x8e),   // NOT #ff007c for text (contrast)
    Success => (0x73,0xda,0xca),
    Info => (0x89,0xdd,0xff),
    Warn => (0xe0,0xaf,0x68),
    // …
  }
}
fn paint_slot(s: &str, slot: Slot) -> String {
  if no_color() { return s.into() }
  if truecolor() { // COLORTERM=truecolor|24bit
    let (r,g,b)=rgb(slot);
    return format!("\x1b[38;2;{r};{g};{b}m{s}\x1b[0m")
  }
  // 256 fallback: map to nearest 38;5;N, else 16 ANSI foundation
  let code = ansi256(slot); // e.g. cyan → 38;5;44
  format!("\x1b[{code}m{s}\x1b[0m")
}
// status_paint then becomes:
Idle => paint_slot(s, Slot::Success),     // mint #73daca
Working => paint_slot(s, Slot::Info),     // ice #89ddff (was 36 cyan)
Blocked => paint_slot(s, Slot::Error),    // #f7768e bold
Done => paint_slot(s, Slot::FgMuted),
Keep 16 ANSI as foundation: Ghostty maps palette 2/6/4 correctly — but truecolor branch guarantees neon-on-void on *any* terminal (skill §4 Detection hierarchy).

// crates/toron-cli/src/theme.rs — single source of truth
pub struct HackerNight;
impl HackerNight {
  pub const VOID: Color = Color::Rgb(0x08,0x08,0x0e);
  pub const FG: Color = Color::Rgb(0xc8,0xd0,0xe8);
  pub const MUTED: Color = Color::Rgb(0x56,0x5f,0x89);
  pub const CYAN: Color = Color::Rgb(0x0d,0xb9,0xd7);
  pub const LAV: Color = Color::Rgb(0xbb,0x9a,0xf7);
  pub const MINT: Color = Color::Rgb(0x73,0xda,0xca);
  pub const AMBER: Color = Color::Rgb(0xe0,0xaf,0x68);
  pub const RED: Color = Color::Rgb(0xf7,0x76,0x8e);
  pub const ICE: Color = Color::Rgb(0x89,0xdd,0xff);
  // semantic Styles
  pub fn header() -> Style {
    Style::default().fg(Self::FG).bg(Self::VOID)
      .add_modifier(Modifier::BOLD)
  }
  pub fn selection() -> Style {
    Style::default().fg(Self::FG).bg(Color::Rgb(0x24,0x24,0x3f))
      .add_modifier(Modifier::BOLD)
  }
  pub fn muted() -> Style {
    Style::default().fg(Self::MUTED)
      .add_modifier(Modifier::DIM)
  }
  pub fn error() -> Style {
    Style::default().fg(Self::RED)
      .add_modifier(Modifier::BOLD)
  }
}
// usage in tui.rs:
let tabs = Tabs::new(TABS).select(state.tab)
  .style(HackerNight::muted())
  .highlight_style(HackerNight::header()
    .fg(HackerNight::CYAN)
    .add_modifier(Modifier::UNDERLINED));
// respect NO_COLOR at startup:
if std::env::var_os("NO_COLOR").is_some() {
  // downgrade every Style to .fg(Color::Reset)
}
